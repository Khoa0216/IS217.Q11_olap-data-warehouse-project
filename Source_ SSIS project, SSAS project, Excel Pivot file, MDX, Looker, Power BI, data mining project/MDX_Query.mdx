//CÁCH 1

FROM [TMDB Movie Data Ware House];

//CÁCH 2
WITH
  MEMBER [Measures].[Aaverage_Revenue] AS
    [Measures].[REVENUE] /
    IIF(
      [Measures].[FACT MOVIE Count] = 0,
      NULL,
      [Measures].[FACT MOVIE Count]
    )
SELECT
  { [Measures].[Aaverage_Revenue] } ON COLUMNS, NON EMPTY [DIM DATE].[Decade].MEMBERS ON ROWS
FROM [TMDB Movie Data Ware House];

--QUERY 2--
//Liệt kê top 15 bộ phim hành động có độ phổ biến (popularity) cao nhất theo từ năm 2020-2025
SELECT 
    {[Measures].[POPULARITY]} ON COLUMNS,
    TOPCOUNT(
        NONEMPTY(
            [DIM MOVIE].[TITLE].[TITLE].MEMBERS,
            ([DIM GENRE].[GENRE NAME].[Action],
             [DIM DATE].[YEAR].&[2020]&[2020]:[DIM DATE].[YEAR].&[2025]&[2020],
             [Measures].[POPULARITY])
        ),
        15,
        [Measures].[POPULARITY]
    ) ON ROWS
FROM [TMDB Movie Data Ware House];

--QUERY 3--
//Phân tích tỉ lệ sinh lời theo thể loại phim trong năm 2020-2025.
SELECT NON EMPTY [DIM GENRE].[GENRE NAME].MEMBERS ON ROWS, [Measures].[Profitability] ON COLUMNS
FROM [TMDB Movie Data Ware House]
WHERE { [DIM DATE].[YEAR].&[2020]&[2020] : [DIM DATE].[YEAR].&[2025]&[2020] };

--QUERY 4--
//Tính tỷ lệ phim có gắn nhãn “adult” theo từng thập kỷ, drill down từ thập kỷ xuống năm ở những thập kỷ 2000, 2010, 2020.
WITH 
 MEMBER [Measures].[% Adult Films] AS
   IIF(
     [Measures].[FACT MOVIE Count] = 0, 
     NULL, 
     (
       [DIM MOVIE].[Adult].&[True], 
       [Measures].[FACT MOVIE Count]
     ) / [Measures].[FACT MOVIE Count]
   ), FORMAT_STRING = "Percent"

SELECT
  { [Measures].[% Adult Films] } ON COLUMNS,

  NON EMPTY
    DrilldownMember(
      [DIM DATE].[Hierarchy].[Decade],
      {
        [DIM DATE].[Hierarchy].[Decade].&[2000],
        [DIM DATE].[Hierarchy].[Decade].&[2010],
        [DIM DATE].[Hierarchy].[Decade].&[2020]
      }
    )
  ON ROWS
FROM [TMDB Movie Data Ware House];

--QUERY 5--
//Thống kê TOP 5 ngôn ngữ là ngôn ngữ gốc có số lượng phim lớn nhất.
SELECT [Measures].[FACT MOVIE Count] ON COLUMNS,
    TOPCOUNT 
    (
        [DIM LANGUAGE].[LANGUAGE NAME].CHILDREN, 5,
        [Measures].[FACT MOVIE Count]
    ) ON ROWS
FROM [TMDB Movie Data Ware House]
WHERE [BRIDGE MOVIE LANGUAGE].[IS ORIGINAL].&[True];

--QUERY 6--
//Liệt kê những phim (TITLE) thuộc thể loại Action, Horror có lợi nhuận âm theo năm.
WITH
  MEMBER [Measures].[Profit] AS 
    ([Measures].[REVENUE] - [Measures].[BUDGET])
SELECT
  NON EMPTY {[Measures].[REVENUE], [Measures].[BUDGET], [Measures].[Profit]} ON COLUMNS,
  NON EMPTY 
  {
    FILTER (
      FILTER(
        [DIM DATE].[Hierarchy].[YEAR].MEMBERS,
        IsNumeric([DIM DATE].[Hierarchy].CurrentMember.Name) = TRUE
      ) * [DIM MOVIE].[TITLE].MEMBERS,
      [Measures].[Profit] < 0
      AND
      (
        ( [DIM GENRE].[GENRE NAME].&[Action], [Measures].[FACT MOVIE Count] ) > 0
        OR
        ( [DIM GENRE].[GENRE NAME].&[Horror], [Measures].[FACT MOVIE Count] ) > 0
      )
    ) 
  } ON ROWS
FROM [TMDB Movie Data Ware House];

--QUERY 7--
//Thống kê TOP 3 quốc gia có nhiều phim phát hành nhất từ năm 2010.
SELECT
  { [Measures].[FACT MOVIE Count] } ON COLUMNS,
  NON EMPTY
    TOPCOUNT(
      [DIM COUNTRY].[COUNTRY NAME].CHILDREN,
      3,
      [Measures].[FACT MOVIE Count]
    )
  ON ROWS
FROM [TMDB Movie Data Ware House]
WHERE
(
    Filter(
        [DIM DATE].[Hierarchy].[YEAR].MEMBERS,
        IsNumeric([DIM DATE].[Hierarchy].CurrentMember.Name) = TRUE AND
        CSng([DIM DATE].[Hierarchy].CurrentMember.Name) >= 2010
    )
);

--QUERY 8--
//Tìm Top 3 ngôn ngữ có doanh thu trung bình cao nhất.
SELECT
  { [Measures].[Average_Revenue] } ON COLUMNS,
  NON EMPTY
    TOPCOUNT(
      [DIM LANGUAGE].[LANGUAGE NAME].CHILDREN,
      3,
      [Measures].[Average_Revenue]
    )
  ON ROWS
FROM [TMDB Movie Data Ware House];

--QUERY 9--
//Tính TOP 5 quốc gia có tỷ lệ phim sinh lời cao nhất từ năm 2010 đến 2025.
WITH
  MEMBER [Measures].[Profitability Rate] AS
    IIF(
      [Measures].[BUDGET] = 0 OR [Measures].[BUDGET] IS NULL,
      NULL,
      ([Measures].[REVENUE] - [Measures].[BUDGET]) / [Measures].[BUDGET]
    )
SELECT
  { [Measures].[Profitability Rate] } ON COLUMNS,

  NON EMPTY
    TOPCOUNT(
      [DIM COUNTRY].[COUNTRY NAME].CHILDREN,
      5,
      [Measures].[Profitability Rate]
    )
  ON ROWS
FROM [TMDB Movie Data Ware House]
WHERE
  (
    Filter(
      [DIM DATE].[Hierarchy].[YEAR].MEMBERS,
      IsNumeric([DIM DATE].[Hierarchy].CurrentMember.Name) = TRUE AND
      CSng([DIM DATE].[Hierarchy].CurrentMember.Name) >= 2010 AND
      CSng([DIM DATE].[Hierarchy].CurrentMember.Name) <= 2025
    )
  );

--QUERY 10--
//Phân tích tổng ngân sách phim theo thể loại và thập kỷ.
SELECT
  NON EMPTY [DIM DATE].[Hierarchy].[Decade].MEMBERS ON COLUMNS,
  NON EMPTY [DIM GENRE].[GENRE NAME].MEMBERS ON ROWS
FROM [TMDB Movie Data Ware House]
WHERE [Measures].[BUDGET];

--QUERY 11--
// Phân tích Tăng trưởng Doanh thu (Year-over-Year Growth)
WITH
  MEMBER [Measures].[REVENUE - PREVIOUS YEAR] AS
    (
      ParallelPeriod(
        [DIM DATE].[YEAR].[YEAR], 
        1,                          
        [DIM DATE].[YEAR].CurrentMember
      ),
      [Measures].[REVENUE]
    )

  MEMBER [Measures].[REVENUE - YoY Growth %] AS
    IIF(
      [Measures].[REVENUE - PREVIOUS YEAR] = 0,
      NULL, 
      ( [Measures].[REVENUE] - [Measures].[REVENUE - PREVIOUS YEAR] ) 
        / [Measures].[REVENUE - PREVIOUS YEAR]
    ), FORMAT_STRING = "Percent" 

SELECT
  { [Measures].[REVENUE], [Measures].[REVENUE - PREVIOUS YEAR], [Measures].[REVENUE - YoY Growth %] } ON 0,

  NON EMPTY [DIM DATE].[YEAR].[YEAR].MEMBERS ON 1

FROM [TMDB Movie Data Ware House];

 --QUERY 12--
 //Tỷ trọng % đóng góp của các Thể loại (Genre) đã thay đổi như thế nào qua từng Thập kỷ (Decade)?
 //Thể loại nào đang 'hot' lên và thể loại nào đang mất dần thị phần?
WITH
  MEMBER [Measures].[% Share of Decade] AS
    IIF(
      ([Measures].[REVENUE], [DIM GENRE].[GENRE NAME].DefaultMember) = 0,
      NULL, 
      
      [Measures].[REVENUE] / ([Measures].[REVENUE], [DIM GENRE].[GENRE NAME].DefaultMember)
    ), FORMAT_STRING = "Percent"

SELECT
  NON EMPTY [DIM DATE].[Decade].[Decade].MEMBERS ON 0,
  NON EMPTY [DIM GENRE].[GENRE NAME].[GENRE NAME].MEMBERS ON 1
FROM [TMDB Movie Data Ware House]
WHERE
  ([Measures].[% Share of Decade])
 
 --QUERY 13--
 //So sánh Tổng Ngân sách (Budget) và Tổng Doanh thu (Revenue) của 3 Công ty Sản xuất (Production Company) hàng đầu 
 // tính cho các phim thuộc 2 thể loại 'Family' và 'Action' , trong các năm 2000 trở về sau (tức là Thập kỷ 2000, 2010 và 2020)
 SELECT

    { [DIM DATE].[Decade].&[2000],[DIM DATE].[Decade].&[2010],[DIM DATE].[Decade].&[2020] }
  )

  --QUERY 14--
  //Tính Tỷ suất Lợi nhuận (ROI) (Công thức: (Revenue - Budget) / Budget) cho từng Thể loại (Genre)
  WITH
  MEMBER [Measures].[ROI] AS
    IIF(
      [Measures].[BUDGET] = 0, 
      NULL,
      ([Measures].[REVENUE] - [Measures].[BUDGET]) / [Measures].[BUDGET]
    ), FORMAT_STRING = "Percent" 

SELECT
  { [Measures].[BUDGET], [Measures].[REVENUE], [Measures].[ROI] } ON 0,

  NON EMPTY
  Order(
    [DIM GENRE].[GENRE NAME].[GENRE NAME].MEMBERS,
    [Measures].[ROI], 
    BDESC 
  ) ON 1

FROM [TMDB Movie Data Ware House]

--QUERY 15--
--Tìm ra các "Thể loại Đột phá Hiệu quả" (Efficiency Breakout Genres) bằng cách so sánh thập kỷ 2010s với 2000s.
-- Ngân sách (Budget) GIẢM nhưng Tỷ suất sinh lời (ROI) TĂNG.
-- Với mỗi thể loại thỏa mãn, liệt kê Top 3 phim có lợi nhuận cao nhất trong thập kỷ 2010, kèm theo % đóng góp của phim đó vào tổng lợi nhuận của thể loại.
WITH
-- Tính Lợi nhuận (Profit)
MEMBER [Measures].[Profit] AS
    [Measures].[REVENUE] - [Measures].[BUDGET]

-- Tính ROI (Return on Investment)
MEMBER [Measures].[ROI] AS
    IIF([Measures].[BUDGET] = 0, NULL,
    ([Measures].[REVENUE] - [Measures].[BUDGET]) / [Measures].[BUDGET]), FORMAT_STRING = "Percent"

-- Tính chênh lệch ROI giữa thập kỷ 2010 và 2000
MEMBER [Measures].[ROI_Growth_10s_vs_00s] AS
    ([Measures].[ROI], [DIM DATE].[Decade].&[2010]) -
    ([Measures].[ROI], [DIM DATE].[Decade].&[2000])

-- Tính chênh lệch Ngân sách giữa thập kỷ 2010 và 2000
MEMBER [Measures].[Budget_Delta_10s_vs_00s] AS
    ([Measures].[BUDGET], [DIM DATE].[Decade].&[2010]) -
    ([Measures].[BUDGET], [DIM DATE].[Decade].&[2000])

-- Tính % Đóng góp lợi nhuận của phim trong thể loại hiện tại
-- Logic: (Lợi nhuận phim) / (Lợi nhuận tổng của Thể loại đó)
MEMBER [Measures].[Pct_Share_Of_Genre] AS
    IIF(
        ([Measures].[Profit], [DIM MOVIE].[TITLE].DefaultMember) = 0,
        NULL,
        [Measures].[Profit] / ([Measures].[Profit], [DIM MOVIE].[TITLE].DefaultMember)
    ), FORMAT_STRING = "Percent"

-- Tạo tập hợp "Thể loại Đột phá" (ROI tăng, Budget giảm)
SET [Smart_Genres] AS
    Filter(
        [DIM GENRE].[GENRE NAME].[GENRE NAME].MEMBERS,
        [Measures].[ROI_Growth_10s_vs_00s] > 0
        AND
        [Measures].[Budget_Delta_10s_vs_00s] < 0
    )

SELECT
    {
        [Measures].[BUDGET],
        [Measures].[REVENUE],
        [Measures].[Profit],
        [Measures].[ROI],
        [Measures].[Pct_Share_Of_Genre]
    } ON COLUMNS,

    NON EMPTY
    Generate(
        [Smart_Genres],
        -- Kỹ thuật tạo báo cáo dạng cây (Genre -> Movies)
        -- Do Genre và Movie là 2 Dimension khác nhau, ta dùng Crossjoin để ghép cặp
        Union(
            -- Dòng 1: Dòng tổng của Thể loại (Ghép Thể loại với "All Movies")
            Crossjoin(
                {[DIM GENRE].[GENRE NAME].CurrentMember},
                {[DIM MOVIE].[TITLE].DefaultMember} 
            ),
            -- Dòng 2-4: Top 3 Phim thuộc thể loại đó
            Crossjoin(
                {[DIM GENRE].[GENRE NAME].CurrentMember},
                TopCount(
                    NONEMPTY(
                        [DIM MOVIE].[TITLE].[TITLE].MEMBERS,
                        -- Bối cảnh: Phải thuộc Thể loại đang xét và thuộc thập kỷ 2010
                        {[DIM GENRE].[GENRE NAME].CurrentMember} * {[DIM DATE].[Decade].&[2010]}
                    ),
                    3,
                    [Measures].[Profit]
                )
            )
        )
    ) ON ROWS
FROM
    [TMDB Movie Data Ware House]
WHERE
    ([DIM DATE].[Decade].&[2010]) -- Lọc hiển thị số liệu của thập kỷ 2010
